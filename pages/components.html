let totalItems = 150, perPage = 15, currentPage = 1;
const BLOCK = 6; // 한 번에 표시할 페이지 번호 수

const totalPages = () => Math.max(1, Math.ceil(totalItems / perPage));

// 고정 블록: currentPage가 속한 블록의 start/end/blockIndex 반환
function calcBlock(cp, tp) {
  const blockIndex = Math.floor((cp - 1) / BLOCK);
  const start = blockIndex * BLOCK + 1;
  const end   = Math.min(tp, start + BLOCK - 1);
  return { start, end, blockIndex };
}

// 페이지 번호 버튼 동적 렌더링 (블록 고정)
function renderNumbers() {
  const container = document.getElementById('pg-numbers');
  container.innerHTML = '';
  const tp = totalPages();
  const { start, end } = calcBlock(currentPage, tp);
  for (let p = start; p <= end; p++) {
    const btn = document.createElement('button');
    btn.className = 's1-pagination-number' + (p === currentPage ? ' is-selected' : '');
    btn.textContent = p;
    btn.addEventListener('click', () => { currentPage = p; update(); });
    container.appendChild(btn);
  }
}

// 화살표 disabled 상태 및 보기옵션 업데이트
function update() {
  const tp = totalPages(), single = tp <= 1;
  const block = calcBlock(currentPage, tp);
  renderNumbers();
  const demo = document.getElementById('pagination-demo');
  // |< : 맨 처음 페이지 — 1페이지이면 비활성
  demo.querySelector('[data-pg-action="first"]').disabled = currentPage === 1 || single;
  // <  : 이전 블록 — 첫 번째 블록이면 비활성
  demo.querySelector('[data-pg-action="prev"]').disabled  = block.blockIndex === 0 || single;
  // >  : 다음 블록 — 마지막 블록이면 비활성
  demo.querySelector('[data-pg-action="next"]').disabled  = block.end >= tp || single;
  // >| : 맨 마지막 페이지 — 마지막 페이지이면 비활성
  demo.querySelector('[data-pg-action="last"]').disabled  = currentPage === tp || single;
  document.getElementById('pg-view-btn').disabled = single;
}

// 이동 버튼 이벤트
const demo = document.getElementById('pagination-demo');
// |< 첫 페이지로
demo.querySelector('[data-pg-action="first"]').addEventListener('click', () => { currentPage = 1; update(); });
// < 이전 블록의 첫 페이지로
demo.querySelector('[data-pg-action="prev"]').addEventListener('click', () => {
  const block = calcBlock(currentPage, totalPages());
  if (block.blockIndex > 0) { currentPage = (block.blockIndex - 1) * BLOCK + 1; update(); }
});
// > 다음 블록의 첫 페이지로
demo.querySelector('[data-pg-action="next"]').addEventListener('click', () => {
  const tp = totalPages();
  const block = calcBlock(currentPage, tp);
  const nextStart = block.end + 1;
  if (nextStart <= tp) { currentPage = nextStart; update(); }
});
// >| 마지막 페이지로
demo.querySelector('[data-pg-action="last"]').addEventListener('click', () => { currentPage = totalPages(); update(); });

// 보기 옵션 드롭다운 (15 / 30 / 50 / 100개씩 보기)
const viewBtn = document.getElementById('pg-view-btn');
const viewDD  = document.getElementById('pg-view-dropdown');

viewBtn.addEventListener('click', () => {
  viewDD.hasAttribute('hidden') ? viewDD.removeAttribute('hidden') : viewDD.setAttribute('hidden', '');
});
viewDD.addEventListener('click', e => {
  const item = e.target.closest('[data-pg-view]');
  if (!item) return;
  perPage = parseInt(item.dataset.pgView, 10);
  document.getElementById('pg-view-label').textContent = perPage + '개씩 보기';
  viewDD.querySelectorAll('[data-pg-view]').forEach(el => el.classList.toggle('is-selected', el === item));
  viewDD.setAttribute('hidden', '');
  currentPage = 1;
  update();
});
document.addEventListener('click', e => {
  if (!document.getElementById('pg-view-wrapper').contains(e.target))
    viewDD.setAttribute('hidden', '');
});

update(); // 초기화
